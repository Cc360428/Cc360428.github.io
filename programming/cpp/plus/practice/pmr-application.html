C++PMR怎样构造真正有界的失败案例 ​

C++PMR怎样构造真正有界的失败案例：把null_memory_resource传作上游，一个槽首次申请成功，第二次相同申请必须抛bad_alloc，第一块与计数保持有效。默认new_delete_resource允许额外内存，只有池本身有界；不能把默认资源的总内存宣称为固定上界。测试避免请求天文数字赌系统OOM。

# Fixed slab / PMR integrated teaching project

示例未执行。C11 slab 库被 C++20 memory_resource 适配器、PMR list 客户端和两个测试入口共同使用，无第三方依赖。单线程；不是通用 malloc 替代品。

输入在 config.hpp：256 字节、8 块、整数1到6；预期 allocator_demo 输出 `sum=21 live=0` 后换行，退出0。list 节点布局依标准库变化，不能承诺6次都命中池；上游保证支持更大节点。默认 new_delete_resource 允许溢出，因此整个适配器不是有界内存；用 null_memory_resource 才禁止上游成功分配。

手动步骤仅供读者，本文未执行：Windows PowerShell、Linux Bash、macOS Zsh 均在本目录执行 `cmake -S . -B build`，`cmake --build build --config Debug`，`ctest --test-dir build -C Debug --output-on-failure`。单配置产物在 build，多配置通常在 build/Debug；执行对应 allocator_demo 或 allocator_demo.exe。需预先具备 CMake >=3.20、兼容C11/C++20工具链及PMR标准库，不在站点安装或运行。

slab_create 两次 malloc；初始化失败不发布句柄。池空与不支持的大小/对齐走上游，释放按池块起点等值归属决定路径，不能仅根据大小重算，因为池空时小请求也会走上游。分配与释放的 bytes/alignment 必须相同，指针必须来自当前等价资源且尚未释放。资源先构造、容器后构造，容器先销毁。析构检查未归还计数并 terminate 是教学诊断，不代替调用契约。

所有 C 句柄禁止按值复制、并发访问、外部修改元数据。malloc backing 按 max_align_t 对齐，stride向上取整；不承诺扩展对齐。free list 索引存外部元数据，位图记录当前借出；重复归还只在未重新分配期间可检测。无法判断伪造、悬空指针或复用后旧别名；调用者必须保证传入的指针值可合法求值。slab_give 不销毁C++对象，容器负责先析构元素再释放存储。

测试是完整源码，TEST-SPEC 列出预期和未实现故障注入。没有声称 native AST、编译、CMake、测试、性能或三平台运行成功。
slab.c（同源文件，示例未执行）
c
#include "slab.h"
#include <stdint.h>
#include <stdlib.h>
struct slab {
    unsigned char *storage;
    size_t width, stride, count, head, live;
    size_t next[64];
    uint64_t used;
};
size_t slab_alignment(void) { return _Alignof(max_align_t); }
slab *slab_create(size_t width, size_t count) {
    size_t a = slab_alignment(), stride;
    slab *s;
    if (!width || !count || count > 64 || width > SIZE_MAX - (a - 1)) return NULL;
    stride = ((width + a - 1) / a) * a;
    if (stride > SIZE_MAX / count) return NULL;
    s = malloc(sizeof *s);
    if (!s) return NULL;
    s->storage = malloc(stride * count);
    if (!s->storage) { free(s); return NULL; }
    s->width = width; s->stride = stride; s->count = count;
    s->head = 0; s->live = 0; s->used = 0;
    for (size_t i = 0; i < count; ++i) s->next[i] = i + 1;
    return s;
}
void *slab_take(slab *s, size_t bytes, size_t a) {
    size_t i;
    if (!s || !bytes || !a || (a & (a - 1)) ||
        bytes > s->width || slab_alignment() % a || s->head == s->count) return NULL;
    i = s->head; s->head = s->next[i];
    s->used |= UINT64_C(1) << i; ++s->live;
    return s->storage + i * s->stride;
}
static size_t locate(const slab *s, const void *q) {
    /* Equality only; no subtraction/order comparison with unrelated pointers. */
    for (size_t i = 0; i < s->count; ++i)
        if (q == (const void *)(s->storage + i * s->stride)) return i;
    return s->count;
}
int slab_owns(const slab *s, const void *q) {
    return s && q && locate(s, q) != s->count;
}
int slab_give(slab *s, void *q) {
    size_t i;
    if (!s || !q) return -1;
    i = locate(s, q);
    if (i == s->count || !(s->used & (UINT64_C(1) << i))) return -1;
    s->used &= ~(UINT64_C(1) << i);
    s->next[i] = s->head; s->head = i; --s->live;
    return 0;
}
size_t slab_live(const slab *s) { return s ? s->live : 0; }
int slab_destroy(slab *s) {
    if (!s) return 0;
    if (s->live) return -1;
    free(s->storage); free(s); return 0;
}

extern "C" {

typedef struct slab slab;
/* Single-thread only. count in [1,64], block_bytes > 0.
   Owns malloc storage; supports fundamental alignment only. */
slab *slab_create(size_t block_bytes, size_t count);
/* Returns NULL for unsupported request or exhaustion; never changes state on failure. */
void *slab_take(slab *, size_t bytes, size_t alignment);
/* q must be NULL or a determinate pointer value to still-live storage.
   Rejects non-block starts and duplicate returns while the slab exists.
   Does NOT validate dangling/forged pointers or detect stale aliases after reuse. */

int slab_owns(const slab *, const void *q);
size_t slab_live(const slab *);
size_t slab_alignment(void);
/* Busy returns -1 without freeing. NULL is accepted. */
int slab_destroy(slab *);

slab_resource.hpp（同源文件，示例未执行）
cpp
#pragma once
#include "slab.h"
#include <cstddef>
#include <exception>
#include <memory_resource>
#include <new>
#include <stdexcept>
class slab_resource final : public std::pmr::memory_resource {
    slab *pool_;
    std::pmr::memory_resource *upstream_;
    std::size_t upstream_live_ = 0;
    void *do_allocate(std::size_t bytes, std::size_t alignment) override {
        if (void *p = slab_take(pool_, bytes, alignment)) return p;
        void *p = upstream_->allocate(bytes, alignment);
        ++upstream_live_; // counts successful upstream allocations only
        return p;
    }
    void do_deallocate(void *p, std::size_t bytes, std::size_t alignment) override {
        // Preconditions: original pointer, size and alignment; not arbitrary pointer validation.
        if (slab_owns(pool_, p)) {
            if (slab_give(pool_, p) != 0) std::terminate();
        } else {
            upstream_->deallocate(p, bytes, alignment);
            --upstream_live_;
        }
    }
    bool do_is_equal(const std::pmr::memory_resource& other) const noexcept override {
        return this == &other;
    }
public:
    explicit slab_resource(std::size_t width, std::size_t count,
                           std::pmr::memory_resource *upstream = std::pmr::new_delete_resource())
        : pool_(nullptr), upstream_(upstream) {
        if (!upstream) throw std::invalid_argument("null upstream");
        pool_ = slab_create(width, count);
        if (!pool_) throw std::bad_alloc();
    }
    slab_resource(const slab_resource&) = delete;
    slab_resource& operator=(const slab_resource&) = delete;
    ~slab_resource() override {
        if (upstream_live_ || slab_destroy(pool_) != 0) std::terminate();
    }
    std::size_t live() const noexcept { return slab_live(pool_) + upstream_live_; }
};

test_resource.cpp（同源文件，示例未执行）
cpp
#include "slab_resource.hpp"
#include <list>
#include <type_traits>
#define CHECK(x) do { if (!(x)) return __LINE__; } while (0)
class tracking final : public std::pmr::memory_resource {
    void *do_allocate(std::size_t n, std::size_t a) override {
        void *p = std::pmr::new_delete_resource()->allocate(n, a); ++live; return p;
    }
    void do_deallocate(void *p, std::size_t n, std::size_t a) override {
        std::pmr::new_delete_resource()->deallocate(p, n, a); --live;
    }
    bool do_is_equal(const std::pmr::memory_resource& r) const noexcept override { return this == &r; }
public: int live = 0;
};
int main() {
    tracking upstream;
    slab_resource r(256, 1, &upstream), other(256, 1, &upstream);
    CHECK(r.is_equal(r) && !r.is_equal(other));
    void *p = r.allocate(16, 1), *q = r.allocate(16, 1);
    CHECK(upstream.live == 1); // exhaustion forwards even supported request
    r.deallocate(q, 16, 1); r.deallocate(p, 16, 1);
    void *large = r.allocate(1024, 1);
    CHECK(upstream.live == 1); r.deallocate(large, 1024, 1);
    void *over = r.allocate(32, slab_alignment() * 2);
    CHECK(upstream.live == 1); r.deallocate(over, 32, slab_alignment() * 2);
    {
        std::pmr::list<int> a(&r); a.push_back(7);
        std::pmr::list<int> explicit_copy(a, &other);
        CHECK(explicit_copy.get_allocator().resource() == &other);
        CHECK(explicit_copy.front() == 7);
    }
    slab_resource bounded(16, 1, std::pmr::null_memory_resource());
    void *only = bounded.allocate(1, 1);
    bool failed = false;
    try { (void)bounded.allocate(1, 1); } catch (const std::bad_alloc&) { failed = true; }
    CHECK(failed && bounded.live() == 1); bounded.deallocate(only, 1, 1);
    CHECK(upstream.live == 0 && r.live() == 0 && other.live() == 0);
    static_assert(!std::allocator_traits<std::pmr::polymorphic_allocator<int>>::propagate_on_container_move_assignment::value);
    return 0;
}

独立练习 ​

将默认上游换成null资源后，如何使失败与标准库节点尺寸无关？

参考答案（先独立推演）

直接对resource申请1字节、1对齐，单槽先占满再申请第二次，捕获bad_alloc并检查首块仍可归还；不要依赖list内部节点大小。
