tasks/tests.py ​

py
from django.contrib.auth import get_user_model
from django.test import Client, TestCase
from django.urls import reverse
from .models import Task

class TaskboardTests(TestCase):
    @classmethod
    def setUpTestData(cls):
        cls.alice = get_user_model().objects.create_user(username="alice", password="local-test-password")
        cls.bob = get_user_model().objects.create_user(username="bob", password="local-test-password")
        cls.foreign = Task.objects.create(owner=cls.bob, title="Bob private")

    def setUp(self):
        self.client.force_login(self.alice)

    def test_anonymous_redirects(self):
        self.client.logout()
        response = self.client.get(reverse("tasks:list"))
        self.assertEqual(response.status_code, 302)
        self.assertIn(reverse("login"), response.url)

    def test_crud_and_filters(self):
        response = self.client.post(reverse("tasks:create"), {"title": "Read ORM"})
        self.assertRedirects(response, reverse("tasks:list"))
        task = Task.objects.get(owner=self.alice)
        self.assertContains(self.client.get(reverse("tasks:list")), "Read ORM")
        self.assertNotContains(self.client.get(reverse("tasks:list")), "Bob private")
        self.client.post(reverse("tasks:edit", args=[task.pk]), {"title": "Read forms", "done": "on"})
        task.refresh_from_db()
        self.assertTrue(task.done)
        self.assertEqual(task.title, "Read forms")
        self.assertNotContains(self.client.get(reverse("tasks:list"), {"state": "open"}), "Read forms")
        self.assertContains(self.client.get(reverse("tasks:list"), {"state": "done"}), "Read forms")
        self.assertEqual(self.client.get(reverse("tasks:delete", args=[task.pk])).status_code, 405)
        self.assertRedirects(self.client.post(reverse("tasks:delete", args=[task.pk])), reverse("tasks:list"))
        self.assertFalse(Task.objects.filter(pk=task.pk).exists())

    def test_other_owner_cannot_read_edit_delete(self):
        for name in ["tasks:edit", "tasks:delete"]:
            url = reverse(name, args=[self.foreign.pk])
            self.assertEqual(self.client.post(url, {"title": "stolen"}).status_code, 404)
        self.assertEqual(self.client.get(reverse("tasks:edit", args=[self.foreign.pk])).status_code, 404)
        self.foreign.refresh_from_db()
        self.assertEqual(self.foreign.title, "Bob private")

    def test_invalid_duplicate_and_forged_owner(self):
        url = reverse("tasks:create")
        self.assertEqual(self.client.post(url, {}).status_code, 200)
        for title in ["   ", "x" * 121]:
            self.assertEqual(self.client.post(url, {"title": title}).status_code, 200)
        self.assertEqual(Task.objects.filter(owner=self.alice).count(), 0)
        self.client.post(url, {"title": "Unique", "owner": self.bob.pk})
        self.assertTrue(Task.objects.filter(title="Unique", owner=self.alice).exists())
        self.assertContains(self.client.post(url, {"title": "Unique"}), "保存冲突")
        self.assertEqual(Task.objects.filter(owner=self.alice).count(), 1)
        self.assertEqual(self.client.get(reverse("tasks:list")).status_code, 200)
        task = Task.objects.get(owner=self.alice, title="Unique")
        edit_url = reverse("tasks:edit", args=[task.pk])
        self.assertRedirects(self.client.post(edit_url, {"title": "Unique"}), reverse("tasks:list"))
        Task.objects.create(owner=self.bob, title="Unique")
        self.assertRedirects(self.client.post(edit_url, {"title": "Unique"}), reverse("tasks:list"))
        Task.objects.create(owner=self.alice, title="Taken")
        self.assertContains(self.client.post(edit_url, {"title": "Taken"}), "保存冲突")
        task.refresh_from_db()
        self.assertEqual(task.title, "Unique")
        self.assertEqual(Task.objects.filter(owner=self.alice).count(), 2)
        self.assertEqual(self.client.get(reverse("tasks:list")).status_code, 200)

    def test_csrf_rejected_and_valid_token_accepted(self):
        client = Client(enforce_csrf_checks=True)
        client.force_login(self.alice)
        url = reverse("tasks:create")
        self.assertEqual(client.post(url, {"title": "No token"}).status_code, 403)
        self.assertFalse(Task.objects.filter(title="No token").exists())
        client.get(url)
        token = client.cookies["csrftoken"].value
        self.assertEqual(client.post(url, {"title": "With token", "csrfmiddlewaretoken": token}).status_code, 302)
        self.assertTrue(Task.objects.filter(title="With token", owner=self.alice).exists())

    def test_login_and_logout_methods(self):
        self.client.logout()
        self.assertTrue(self.client.login(username="alice", password="local-test-password"))
        self.assertEqual(self.client.get(reverse("logout")).status_code, 405)
        self.assertRedirects(self.client.post(reverse("logout")), reverse("login"))

README.md ​

md
# Django 5.2 个人任务板

示例未执行。交付仅做 AST、文件引用、Markdown/Vue 模板和测试规格静态检查，没有安装 Django，没有创建数据库或启动服务。目标 Python 3.12、Django 5.2.18、SQLite；后续运行前自行核对补丁维护状态。

功能：内置用户登录与 POST 退出，个人任务新增、列表、完成状态筛选、编辑和 POST 删除；所有读写通过 owner 过滤；标题必填、最多120字符，同一用户标题唯一。后台仅用于可信管理员；没有开放注册或多租户管理员隔离。重复标题返回表单错误，不保存第二条。

`.env.example` 仅示意变量，代码不自动读取 `.env`。`migrate` 会创建本目录 `db.sqlite3`，迁移源文件手写，需自行用 `makemigrations --check --dry-run` 核对漂移。开发服务器只绑定回环地址。

浏览器打开 http://127.0.0.1:8000/ 应跳转登录；使用创建的用户登录，然后新增“Read ORM”，编辑为完成，筛选完成列表并删除。可通过 admin 的 Users 创建第二个普通用户（设置密码），两用户互不可见任务。管理员后台有跨用户管理能力，不等同普通任务页。

预期测试：六个测试方法全部成功；覆盖 CRUD、筛选、匿名、所有权、空POST/空白/超长、新增和编辑重复冲突、保留原标题与跨用户同名、伪造 owner、CSRF 缺失及有效 token、退出方法限制。普通 TestCase 的客户端默认不强制 CSRF，因此专门建立 enforce_csrf_checks 客户端。测试数据库由 Django 测试运行器管理；本轮未创建。

这不是生产部署配置：部署前提供 WSGI/ASGI 宿主、静态资源、HTTPS、Cookie 策略、数据库备份和密钥管理，运行部署检查并补齐部署环境测试。本工程未验证并发容量、跨平台运行或真实数据库迁移。
